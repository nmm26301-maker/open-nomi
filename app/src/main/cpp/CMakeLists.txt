cmake_minimum_required(VERSION 3.22.1)
project(opennomi_hymt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
  set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
if(ANDROID AND ANDROID_ABI STREQUAL "x86_64")
  set(GGML_AVX OFF CACHE BOOL "" FORCE)
  set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
  set(GGML_FMA OFF CACHE BOOL "" FORCE)
  set(GGML_F16C OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/llama.cpp" "llama-build")
add_library(hymt_core STATIC hymt_core.cpp)
target_link_libraries(hymt_core PUBLIC llama)
target_include_directories(hymt_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
if(ANDROID)
  add_library(opennomi_hymt SHARED hymt_jni.cpp)
  target_link_libraries(opennomi_hymt PRIVATE hymt_core log)
  target_link_options(opennomi_hymt PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_executable(opennomi-hymt-probe hymt_probe.cpp)
  target_link_libraries(opennomi-hymt-probe PRIVATE hymt_core)
endif()
